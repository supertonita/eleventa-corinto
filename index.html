function importarExcelProductos(e) { const file = e.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = async function(evt) { try { const data = new Uint8Array(evt.target.result); const workbook = XLSX.read(data, { type: 'array', raw: false }); // raw: false para leer strings limpios const sheetName = workbook.SheetNames[0]; const rows = XLSX.utils.sheet_to_json(workbook.Sheets[sheetName], { defval: "" }); if (rows.length === 0) { alert("El archivo Excel parece estar vacío."); return; } let importadosCount = 0; let omitidosCount = 0; for (let i = 0; i < rows.length; i++) { const r = rows[i]; // Detección flexible de columnas sin importar mayúsculas/minúsculas o tildes const codigoRaw = String( r.Codigo || r.CÓDIGO || r.codigo || r.CODIGO || r.Barcode || r.BARCODE || r.BARCODE_ID || "" ).trim(); const descripcion = String( r.Descripcion || r.DESCRIPCIÓN || r.descripcion || r.DESCRIPCION || r.Producto || r.PRODUCTO || r.Nombre || "" ).trim(); if (!codigoRaw || !descripcion) { omitidosCount++; continue; } const primerCodigo = codigoRaw.split(',')[0].trim(); const costo = parseFloat(String(r.Costo || r.COSTO || r.costo || 0).replace('$', '').replace(',', '')) || 0; const precio = parseFloat(String(r.Precio || r.PRECIO || r.PrecioVenta || r.VENTA || r.precio || 0).replace('$', '').replace(',', '')) || 0; const departamento = String(r.Departamento || r.DEPARTAMENTO || r.depto || 'Abarrotes').trim(); const unidad = String(r.Unidad || r.UNIDAD || r.Medida || 'LB').trim(); const existencia = parseFloat(r.Existencia || r.EXISTENCIA || r.Stock || 10) || 10; const nuevo = { codigo: codigoRaw, descripcion, costo, precio, departamento, unidad, tipo: 'granel', existencia }; // Actualizar o agregar en la lista local de productos const idx = productos.findIndex(p => productoCoincideConCodigo(p, primerCodigo)); if (idx > -1) { productos[idx] = nuevo; } else { productos.push(nuevo); } // Sincronizar en la nube (Firestore Corinto) if (window.db && window.setDoc && window.doc) { try { await window.setDoc(window.doc(window.db, "productos", primerCodigo), nuevo); } catch (errDb) { console.warn(`Error al subir a nube producto ${primerCodigo}:`, errDb); } } importadosCount++; } // Guardar caché local en Corinto localStorage.setItem('pos_productos_cache_corinto', JSON.stringify(productos)); alert(`¡Proceso completado!\n\n✅ Se cargaron exitosamente: ${importadosCount} productos.\n⚠️ Se omitieron por falta de código o nombre: ${omitidosCount} filas.`); renderTablaCatalogo(); } catch (err) { console.error("Error al procesar el Excel:", err); alert("Ocurrió un error al leer el archivo Excel. Asegúrate de que sea un archivo .xlsx o .csv válido."); } }; reader.readAsArrayBuffer(file); }
